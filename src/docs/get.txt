1. GET — plantilla completa

Incluye los dos casos que vas a encontrar normalmente: GET sin parámetros y GET con parámetro en URL.

# ============================================================
# GET
# ============================================================
#
# GET se utiliza normalmente para OBTENER información.
#
# Generalmente NO recibe requestBody.
#
# Puede recibir:
#
# 1. Sin parámetros:
#
#    router.get("/", ...)
#
#    GET /api/category
#
#
# 2. Parámetros en la URL:
#
#    router.get("/:uuid/category", ...)
#
#    Express:
#    GET /api/category/:uuid/category
#
#    OpenAPI:
#    GET /api/category/{uuid}/category
#
#    En el backend:
#    req.params.uuid
#
#
# 3. Query params:
#
#    GET /api/category?page=1&limit=10
#
#    En el backend:
#    req.query.page
#    req.query.limit
#
#
# REGLA:
#
# req.params.uuid
# -> in: path
#
# req.query.page
# -> in: query
#
# ============================================================


paths:

  # ==========================================================
  # GET SIN PARÁMETROS DE PATH
  # ==========================================================

  /api/category:

    get:

      # Grupo donde aparecerá en Swagger UI.
      tags:
        - Category

      # Nombre corto del endpoint.
      summary: Obtener categorías principales

      # Explicación opcional más detallada.
      description: Obtiene todas las categorías principales registradas.

      # El endpoint necesita JWT.
      # Eliminar si la ruta es pública.
      security:
        - bearerAuth: []


      # -------------------------------------------------------
      # QUERY PARAMS
      # -------------------------------------------------------
      #
      # Si NO utilizas query params, elimina completamente
      # esta sección "parameters".
      #
      # Ejemplo:
      #
      # GET /api/category?page=1&limit=10
      #

      parameters:

        - name: page

          # Significa:
          # req.query.page
          in: query

          # false porque el endpoint puede funcionar sin enviarlo.
          required: false

          description: Número de página solicitada

          schema:
            type: integer
            minimum: 1
            default: 1


        - name: limit

          # Significa:
          # req.query.limit
          in: query

          required: false

          description: Cantidad de registros por página

          schema:
            type: integer
            minimum: 1
            default: 10


      # -------------------------------------------------------
      # RESPUESTAS
      # -------------------------------------------------------

      responses:

        # RESPUESTA EXITOSA
        "200":
          description: Categorías obtenidas correctamente

          content:
            application/json:

              schema:

                # La API devuelve:
                #
                # [
                #   {...},
                #   {...}
                # ]
                #
                # Por eso usamos array.
                type: array

                # Describe cada elemento del array.
                items:

                  # Utiliza el schema Category definido en:
                  #
                  # components:
                  #   schemas:
                  #     Category:
                  #
                  $ref: "#/components/schemas/Category"


        # DATOS / QUERY PARAMS INVÁLIDOS
        "400":
          description: Parámetros inválidos


        # NO AUTENTICADO
        "401":
          description: No autorizado

          content:
            application/json:
              schema:
                $ref: "#/components/schemas/UnauthorizedResponse"


        # ERROR INTERNO
        "500":
          description: Error interno del servidor



  # ==========================================================
  # GET CON PARÁMETRO EN LA URL
  # ==========================================================
  #
  # Express:
  #
  # router.get("/:uuid/category", ...)
  #
  # Ruta:
  #
  # /api/category/:uuid/category
  #
  # OpenAPI usa:
  #
  # /api/category/{uuid}/category
  #
  # ==========================================================

  /api/category/{uuid}/category:

    get:

      tags:
        - Category

      summary: Obtener una categoría por UUID

      description: Obtiene una categoría específica utilizando su UUID.

      security:
        - bearerAuth: []


      # -------------------------------------------------------
      # PARÁMETROS DE LA URL
      # -------------------------------------------------------

      parameters:

        - name: uuid

          # Representa:
          #
          # req.params.uuid
          #
          in: path

          # Todos los parámetros que forman parte del path
          # deben ser obligatorios.
          required: true

          description: UUID de la categoría

          schema:
            type: string
            format: uuid

          example: "4ab89809-d343-4470-b241-daf85ab0f014"


      # -------------------------------------------------------
      # RESPUESTAS
      # -------------------------------------------------------

      responses:

        "200":
          description: Categoría obtenida correctamente

          content:
            application/json:

              schema:

                # Como devuelve UN objeto y no un array,
                # utilizamos Category directamente.
                $ref: "#/components/schemas/Category"


        "400":
          description: UUID inválido


        "401":
          description: No autorizado

          content:
            application/json:
              schema:
                $ref: "#/components/schemas/UnauthorizedResponse"


        "404":
          description: La categoría no existe

          content:
            application/json:
              schema:
                $ref: "#/components/schemas/ErrorResponse"


        "500":
          description: Error interno del servidor
